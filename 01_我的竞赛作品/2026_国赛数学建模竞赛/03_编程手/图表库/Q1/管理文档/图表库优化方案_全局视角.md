#### 任务 I.3：创建一键生成脚本
**目标**：论文手可以一键重新生成所有图表（例如修改配色后）

**文件位置**：`01_生图代码/generate_all.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
一键生成 Q1 所有图表

用法：
    cd 01_生图代码
    python generate_all.py
"""

import sys
import os

# 确保可以导入 q1_figures_beautified
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

from q1_figures_beautified import (
    fig_5_1_1_baseline,
    fig_5_1_2_wedge_evolution,
    fig_5_1_3_jung_counterexample,
    fig_5_1_4_jung_sandwich,
    fig_5_1_5_vertex_filtering,
    fig_5_1_6_param_scan,
    fig_5_1_7_gamma_distribution,
    fig_5_1_8_n_and_grid,
    fig_5_1_10_geometry_construction,
    fig_5_1_11_sensitivity_triplet,
    fig_5_1_12_param_space_partition,
    fig_5_1_13_phi_D_theory,
)

FIGURE_LIST = [
    ('5.1-1', fig_5_1_1_baseline, '基准算例'),
    ('5.1-2', fig_5_1_2_wedge_evolution, '角域构造演化'),
    ('5.1-3', fig_5_1_3_jung_counterexample, 'Jung反例'),
    ('5.1-4', fig_5_1_4_jung_sandwich, 'Jung夹逼'),
    ('5.1-5', fig_5_1_5_vertex_filtering, '顶点过滤'),
    ('5.1-6', fig_5_1_6_param_scan, '参数扫描'),
    ('5.1-7', fig_5_1_7_gamma_distribution, 'γ分布'),
    ('5.1-8', fig_5_1_8_n_and_grid, 'n扫描+热力图'),
    ('5.1-10', fig_5_1_10_geometry_construction, '交会定位区域构造'),
    ('5.1-11', fig_5_1_11_sensitivity_triplet, '灵敏度三联图'),
    ('5.1-12', fig_5_1_12_param_space_partition, '参数空间分区'),
    ('5.1-13', fig_5_1_13_phi_D_theory, 'φ-D理论验证'),
]

if __name__ == '__main__':
    print("=" * 60)
    print("Q1 图表批量生成工具")
    print("=" * 60)
    
    success = 0
    failed = 0
    
    for fig_id, func, name in FIGURE_LIST:
        print(f"\n[图{fig_id}] {name}...", end=' ')
        try:
            func()
            success += 1
            print("✅")
        except Exception as e:
            failed += 1
            print(f"❌ {e}")
    
    print("\n" + "=" * 60)
    print(f"生成完成：✅ {success} 张成功，❌ {failed} 张失败")
    print("=" * 60)
```
